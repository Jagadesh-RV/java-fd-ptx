class Solution {
    public int missingNumber(int[] nums) {
        int len=nums.length;
        int res=0;
        HashSet<Integer> set=new HashSet<>();
        for(int j=0;j<=len;j++){
            set.add(j);
        }
        for(int i=0;i<len;i++){
            if(set.contains(nums[i])){
                continue;
            }else{
                res=nums[i];
            }
        }
        return res;
    }
}
